/**
 * 1. GOOGLE SHEETS — COCKPIT BUILDER
 */
function buildCockpit() {
  const ss = SpreadsheetApp.create(TP.cockpitName);
  PropertiesService.getScriptProperties().setProperty("COCKPIT_ID", ss.getId());

  const definitions = getSheetDefinitions_();
  const defaultSheet = ss.getSheets()[0];

  definitions.forEach((def, i) => {
    const sh = i === 0 ? defaultSheet.setName(def.name) : ss.insertSheet(def.name);
    sh.clear();

    sh.getRange(1, 1, 1, def.headers.length).setValues([def.headers]);

    if (def.sampleRows.length > 0) {
      sh.getRange(2, 1, def.sampleRows.length, def.headers.length).setValues(def.sampleRows);
    }

    formatSheet_(sh, def.headers.length);
    applyValidations_(sh, def.headers, def.validations || {});
  });

  applyCockpitFormulas_(ss);
  SpreadsheetApp.flush();

  return ss;
}

function formatSheet_(sh, lastCol) {
  sh.setFrozenRows(1);
  sh.getRange(1, 1, 1, lastCol)
    .setFontWeight("bold")
    .setBackground("#1f4e78")
    .setFontColor("#ffffff")
    .setWrap(true);

  sh.getDataRange().setWrap(true);
  sh.autoResizeColumns(1, lastCol);

  try {
    sh.getRange(1, 1, Math.max(1, sh.getLastRow()), lastCol).createFilter();
  } catch (e) {}

  sh.setRowHeight(1, 40);
}

function applyValidations_(sh, headers, validations) {
  Object.keys(validations).forEach(header => {
    const col = headers.indexOf(header) + 1;
    if (col <= 0) return;

    const rule = SpreadsheetApp.newDataValidation()
      .requireValueInList(validations[header], true)
      .setAllowInvalid(false)
      .build();

    sh.getRange(2, col, 500, 1).setDataValidation(rule);
  });
}

function row_(headers, obj) {
  return headers.map(h => obj[h] !== undefined ? obj[h] : "");
}

function addDays_(date, days) {
  const d = new Date(date);
  d.setDate(d.getDate() + days);
  return d;
}
