def call(client, method, path, expected=200, **kwargs):
    response = getattr(client, method)(path, **kwargs)
    REQUEST_LOG.append({"event": "request", "method": method.upper(), "path": path,
                        "expected_status": expected, "actual_status": response.status_code,
                        "response_sha256": hashlib.sha256(response.content).hexdigest()})
    assert response.status_code == expected, (method, path, response.status_code, response.text[:300])
    return response


def query(sql, args=()):
    with sqlite3.connect(main.DB) as connection:
        connection.row_factory = sqlite3.Row
        return [dict(row) for row in connection.execute(sql, args)]


def mutate(sql, args=()):
    # Only this test's synthetic external database; there is no evidence PUT API.
    with sqlite3.connect(main.DB) as connection:
        connection.execute(sql, args)


def historical(tender):
    return query("SELECT * FROM decisions WHERE tender_id=? ORDER BY id", (tender,))


def detail(client, tender):
    return call(client, "get", f"/api/tenders/{tender}").json()


def attest(client, tender):
    return call(client, "post", f"/api/tenders/{tender}/attest-source-scope", json={
        "reviewer": "Independent Synthetic Reviewer",
        "checked_full_document_set": True,
        "note": "Reviewed all synthetic documents and annexes for this independent QA case",
    }).json()


def decide(client, tender, decision="BID", expected=201):
    return call(client, "post", f"/api/tenders/{tender}/decisions", expected, json={
        "decision": decision, "reviewer": "Independent Synthetic Reviewer",
        "rationale": "Synthetic evidence and source scope explicitly checked for QA only",
    }).json()


def upload_no_candidates(client, tender, marker="A"):
    buffer = io.BytesIO()
    canvas = Canvas(buffer, invariant=1)
    canvas.drawString(50, 750, f"Calendar update {marker}: delivery date 2030-01-15.")
    canvas.showPage()
    canvas.save()
    result = call(client, "post", f"/api/tenders/{tender}/upload-pdf", 201,
                  files={"file": (f"synthetic-{marker}.pdf", buffer.getvalue(), "application/pdf")}).json()
    assert result["candidates"] == 0
    return result["source_id"]


def make_ready(client, *, with_decision=True, source_count=0, expires_on=None):
    org = call(client, "post", "/api/organizations", 201,
               json={"name": "Independent Synthetic Organization"}).json()["id"]
    tender = call(client, "post", "/api/tenders", 201, json={
        "organization_id": org, "title": "Independent synthetic decision QA tender",
    }).json()["id"]
    sources = [upload_no_candidates(client, tender, str(i)) for i in range(source_count)]
    text = "Three synthetic projects must be documented"
    requirement = call(client, "post", f"/api/tenders/{tender}/requirements", 201, json={
        "text": text, "source_page": 1, "source_quote": text,
    }).json()["id"]
    call(client, "put", f"/api/requirements/{requirement}/review", json={
        "text": text, "source_page": 1, "source_quote": text, "mandatory": True,
    })
    if sources:
        mutate("UPDATE requirements SET source_id=? WHERE id=?", (sources[0], requirement))
    evidence = call(client, "post", "/api/evidence", 201, json={
        "organization_id": org, "label": "Synthetic signed certificate",
        "reference": "INDEPENDENT-SYNTHETIC-001", "verified": True,
        "verification_note": "Checked synthetic certificate for QA only", "expires_on": expires_on,
    }).json()["id"]
    call(client, "put", f"/api/requirements/{requirement}/status", json={
        "status": "VERIFIED", "evidence_id": evidence, "notes": "Synthetic reviewed evidence",
    })
    attest(client, tender)
    assert detail(client, tender)["compliance"] == "READY_FOR_HUMAN_DECISION"
    if with_decision:
        decide(client, tender)
        assert detail(client, tender)["current_decision"] == "BID"
    return {"org": org, "tender": tender, "req": requirement, "evidence": evidence, "sources": sources}


def assert_stale(snapshot):
    assert snapshot["current_decision"] is None
    assert snapshot["decisions"][0]["is_current"] is False


def test_addendum_invalidates_approval_and_attestation_cannot_resurrect_it(pilot):
    case = make_ready(pilot)
    tender = case["tender"]
    history = historical(tender)
    before = detail(pilot, tender)
    upload_no_candidates(pilot, tender)
    after = detail(pilot, tender)
    assert len(after["requirements"]) == len(before["requirements"])
    assert after["source_scope_fingerprint"] != before["source_scope_fingerprint"]
    assert after["source_scope_verified"] is False
    assert after["compliance"] == "UNRESOLVED"
    assert_stale(after)
    decide(pilot, tender, expected=409)
    assert historical(tender) == history
    attest(pilot, tender)
    reattested = detail(pilot, tender)
    assert reattested["compliance"] == "READY_FOR_HUMAN_DECISION"
    assert_stale(reattested)
    assert historical(tender) == history
    decide(pilot, tender)
    fresh = detail(pilot, tender)
    assert fresh["current_decision"] == "BID"
    assert fresh["decisions"][0]["is_current"] is True
    assert fresh["decisions"][1]["is_current"] is False
    assert historical(tender)[:-1] == history


def test_source_reassociation_requires_new_scope_and_new_decision(pilot):
    case = make_ready(pilot, source_count=2)
    tender = case["tender"]
    before = detail(pilot, tender)
    history = historical(tender)
    mutate("UPDATE requirements SET source_id=? WHERE id=?", (case["sources"][1], case["req"]))
    changed = detail(pilot, tender)
    assert changed["sources"] == before["sources"]
    assert changed["source_scope_fingerprint"] != before["source_scope_fingerprint"]
    assert changed["source_scope_verified"] is False
    assert_stale(changed)
    attest(pilot, tender)
    assert_stale(detail(pilot, tender))
    assert historical(tender) == history
    decide(pilot, tender)
    assert detail(pilot, tender)["current_decision"] == "BID"
    assert historical(tender)[:-1] == history


def test_source_content_change_keeps_prior_snapshot_immutable(pilot):
    case = make_ready(pilot, source_count=1)
    history = historical(case["tender"])
    mutate("UPDATE sources SET sha256=? WHERE id=?", ("f" * 64, case["sources"][0]))
    assert_stale(detail(pilot, case["tender"]))
    attest(pilot, case["tender"])
    assert_stale(detail(pilot, case["tender"]))
    assert historical(case["tender"]) == history

@pytest.mark.parametrize("legacy_kind", ["scope_absent", "baseline_snapshot"])
def test_legacy_snapshots_fail_closed_without_rewriting_them(pilot, legacy_kind):
    case = make_ready(pilot)
    row = historical(case["tender"])[0]
    snapshot = json.loads(row["decision_snapshot"])
    if legacy_kind == "scope_absent":
        snapshot.pop("scope_hash", None)
    else:
        # This exactly limits persisted fields to the original ca819a9 shape.
        original_fields = {"compliance", "mandatory_total", "mandatory_verified", "requirement_ids", "fingerprint", "scope_hash"}
        snapshot = {key: value for key, value in snapshot.items() if key in original_fields}
        # Original requirement-only fingerprint, intentionally excluding source/evidence metadata.
        requirements = detail(pilot, case["tender"])["requirements"]
        keys = ("id", "mandatory", "reviewed", "text", "source_page", "source_quote", "status", "effective_status", "evidence_id", "evidence_issue")
        original_fp = [{key: req[key] for key in keys if key in req} for req in requirements]
        snapshot["fingerprint"] = hashlib.sha256(json.dumps(original_fp, sort_keys=True, ensure_ascii=False).encode()).hexdigest()
    mutate("UPDATE decisions SET decision_snapshot=? WHERE id=?", (json.dumps(snapshot), row["id"]))
    history = historical(case["tender"])
    changed = detail(pilot, case["tender"])
    assert changed["compliance"] == "READY_FOR_HUMAN_DECISION"
    assert_stale(changed)
    assert historical(case["tender"]) == history
    decide(pilot, case["tender"])
    assert detail(pilot, case["tender"])["current_decision"] == "BID"
    assert historical(case["tender"])[:-1] == history


def test_noop_reviews_unrelated_evidence_metrics_exports_preserve_current_bid(pilot):
    case = make_ready(pilot)
    before = detail(pilot, case["tender"])
    history = historical(case["tender"])
    unrelated = call(pilot, "post", "/api/evidence", 201, json={
        "organization_id": case["org"], "label": "Unlinked synthetic certificate",
        "reference": "UNRELATED-ONLY", "verified": True,
        "verification_note": "Synthetic unrelated evidence check",
    }).json()["id"]
    mutate("UPDATE evidence SET label=?, expires_on=?, verified=0 WHERE id=?",
           ("Unlinked renamed record", (date.today() - timedelta(days=1)).isoformat(), unrelated))
    call(pilot, "post", f"/api/tenders/{case['tender']}/metrics", json={
        "quoted_bdt": 12500, "paid_bdt": 0, "actual_minutes": 20, "notes": "Synthetic QA metrics",
    })
    req = before["requirements"][0]
    call(pilot, "put", f"/api/requirements/{case['req']}/review", json={
        "text": req["text"], "mandatory": True, "source_page": req["source_page"], "source_quote": req["source_quote"],
    })
    call(pilot, "put", f"/api/requirements/{case['req']}/status", json={
        "status": "VERIFIED", "evidence_id": case["evidence"], "notes": req["notes"],
    })
    for extension in ("csv", "xlsx", "docx"):
        response = call(pilot, "get", f"/api/tenders/{case['tender']}/export.{extension}")
        assert len(response.content) > 70
    after = detail(pilot, case["tender"])
    assert after["current_decision"] == "BID"
    assert after["decisions"][0]["is_current"] is True
    assert historical(case["tender"]) == history

@pytest.mark.parametrize("prefix", ["=SUM(1,2)", "+SUM(1,2)", "-SUM(1,2)", "@SUM(1,2)"])
@pytest.mark.parametrize("whitespace", ["", "  ", "\t", "\r\n  "])
def test_xlsx_all_user_fields_are_literal_with_prefix_and_whitespace_variants(pilot, prefix, whitespace):
    case = make_ready(pilot)
    payload = whitespace + prefix + " synthetic literal"
    # Preserve leading whitespace in synthetic persisted legacy records. API trims
    # most fields; direct DB writes ensure the export handles those legacy values.
    mutate("UPDATE tenders SET title=? WHERE id=?", (payload, case["tender"]))
    mutate("UPDATE requirements SET text=?,source_quote=?,notes=? WHERE id=?", (payload, payload, payload, case["req"]))
    mutate("UPDATE evidence SET label=? WHERE id=?", (payload, case["evidence"]))
    mutate("UPDATE decisions SET reviewer=?,rationale=? WHERE tender_id=?", (payload, payload, case["tender"]))
    history = historical(case["tender"])
    response = call(pilot, "get", f"/api/tenders/{case['tender']}/export.xlsx")
    workbook = load_workbook(io.BytesIO(response.content), data_only=False)
    cells = [("Compliance Matrix", coordinate) for coordinate in ("B2", "E8", "G8", "H8", "I8")]
    cells += [("Decision History", coordinate) for coordinate in ("B2", "C2")]
    for sheet_name, coordinate in cells:
        cell = workbook[sheet_name][coordinate]
        assert cell.data_type == "s", (sheet_name, coordinate, cell.value, cell.data_type)
        assert cell.value in (payload, "'" + payload), (sheet_name, coordinate, cell.value)
    assert not any(cell.data_type == "f" for sheet in workbook for row in sheet for cell in row)
    assert workbook["Compliance Matrix"]["A8"].data_type == "n"
    assert workbook["Compliance Matrix"]["A8"].value == case["req"]
    assert workbook["Compliance Matrix"]["F8"].data_type == "n"
    assert workbook["Compliance Matrix"]["F8"].value == 1
    assert workbook["Decision History"]["E2"].value == history[0]["decision_snapshot"]
    assert historical(case["tender"]) == history
